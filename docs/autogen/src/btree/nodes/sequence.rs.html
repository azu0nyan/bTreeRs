1//! The [`Sequence`] composite (sequence with memory).
2//!
3//! Sequences run children left-to-right and stop at the first failure (logical
4//! AND). This variant remembers its progress between ticks: a running child is
5//! ticked again on the next tick rather than restarting from the first child.
6//! Compare with [`ReactiveSequence`](super::ReactiveSequence) (restarts every
7//! tick) and [`ProgressiveSequence`](super::ProgressiveSequence) (also keeps a
8//! failing child's position).
9
10use super::halt_all;
11use crate::debug::{record, tick_child};
12use crate::{BehaviorNode, BoxNode, DebugNode, Status};
13
14/// A sequence that remembers its progress between ticks ("sequence with
15/// memory").
16///
17/// Ticks children in order. A child failure restarts the whole sequence and
18/// returns [`Status::Failure`]; a running child is ticked again on the next
19/// tick (progress is kept); when all children succeed the sequence resets and
20/// returns [`Status::Success`]. Mirrors the Scala `Sequence`.
21///
22/// > Note: the Scala original advanced with `tasks.tail` (a bug that re-ran the
23/// > second child forever); this port advances correctly by index, matching the
24/// > documented "saves progress between ticks" intent.
25pub struct Sequence<D> {
26    children: Vec<BoxNode<D>>,
27    current: usize,
28}
29
30impl<D> Sequence<D> {
31    /// Build a sequence from its children. Use the [`nodes!`](crate::nodes!)
32    /// macro to box heterogeneous node types ergonomically.
33    pub fn new(children: Vec<BoxNode<D>>) -> Self {
34        Self { children, current: 0 }
35    }
36
37    fn reset(&mut self) {
38        self.current = 0;
39        halt_all(&mut self.children);
40    }
41}
42
43impl<D> BehaviorNode<D> for Sequence<D> {
44    fn tick(&mut self, data: &mut D, mut dbg: Option<&mut DebugNode>) -> Status {
45        let status = loop {
46            if self.current >= self.children.len() {
47                self.reset();
48                break Status::Success;
49            }
50            match tick_child(&mut *self.children[self.current], data, &mut dbg) {
51                Status::Failure => {
52                    self.reset();
53                    break Status::Failure;
54                }
55                Status::Running => break Status::Running,
56                Status::Success => self.current += 1,
57            }
58        };
59        record(dbg, status, || self.node_info());
60        status
61    }
62
63    fn halt(&mut self) {
64        self.reset();
65    }
66
67    fn node_info(&self) -> String {
68        format!("Sequence {} / {}", self.current, self.children.len())
69    }
70}
