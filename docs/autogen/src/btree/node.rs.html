5/// A boxed, type-erased behavior node operating over the context type `D`.
6///
7/// Composite nodes (sequences, decorators, …) store their children as
8/// `BoxNode<D>` so that heterogeneous node types can live in the same tree.
9pub type BoxNode<D> = Box<dyn BehaviorNode<D>>;
10
11/// A node in a behavior tree.
12///
13/// `D` is the *tick data* / context type threaded through the whole tree on
14/// every tick. It is passed by `&mut` so leaf nodes can both read and mutate
15/// the agent / blackboard they act on. The library is fully generic over `D`
16/// and never depends on any engine, so the same tree definition works on a
17/// plain struct in a unit test or on a Godot node in a game.
18///
19/// Implement this trait to add your own custom leaves or composites. The only
20/// required method is [`tick`](BehaviorNode::tick); [`halt`](BehaviorNode::halt)
21/// and [`node_info`](BehaviorNode::node_info) have sensible defaults.
22///
23/// ## Tracing
24///
25/// `tick` takes an optional debug slot, `Option<&mut DebugNode>`. Pass `None`
26/// for the normal zero-overhead path; pass `Some(&mut slot)` to have the node
27/// record a [`DebugNode`] trace of itself and the children it processes. Custom
28/// implementations fill the slot with the [`record`](crate::record) helper
29/// (leaves) and [`tick_child`](crate::tick_child) helper (composites), both of
30/// which no-op when the slot is `None`.
31pub trait BehaviorNode<D> {
32    /// Advance this node by one tick, returning its [`Status`].
33    ///
34    /// A node that returns [`Status::Running`] expects to be ticked again on a
35    /// later frame and should preserve whatever state it needs between calls.
36    ///
37    /// `dbg` is an optional trace slot. When it is `Some`, the node must fill it
38    /// with its own name and status (via [`record`](crate::record)) and, for a
39    /// composite, the traces of every child it ticks (via
40    /// [`tick_child`](crate::tick_child)). When it is `None`, no tracing work is
41    /// done.
42    fn tick(&mut self, data: &mut D, dbg: Option<&mut DebugNode>) -> Status;
43
44    /// Abort a node that is currently [`Status::Running`].
45    ///
46    /// Composite nodes propagate `halt` to their running children so that a
47    /// reactive parent can cleanly cancel a branch it is no longer interested
48    /// in. The default implementation does nothing, which is correct for
49    /// stateless leaves.
50    fn halt(&mut self) {}
51
52    /// A short human-readable description of this node, used when rendering a
53    /// debug trace. The default returns the node's (unqualified) type name;
54    /// stateful nodes override it to surface their progress.
55    fn node_info(&self) -> String {
56        short_type_name::<Self>()
57    }
58
59    /// Tick this node with tracing enabled, returning both the [`Status`] and a
60    /// fully populated [`DebugNode`] trace.
61    ///
62    /// This is just a convenience wrapper around `tick(data, Some(..))`; the
63    /// plain [`tick`](BehaviorNode::tick) with `None` remains the zero-overhead
64    /// path for production use.
65    fn tick_traced(&mut self, data: &mut D) -> (Status, DebugNode) {
66        let mut root = DebugNode::empty();
67        let status = self.tick(data, Some(&mut root));
68        (status, root)
69    }
70}
71
72/// Returns the unqualified type name of `T`, e.g. `Sequence` rather than
73/// `btree::sequence::Sequence<my_game::Ctx>`.
74pub(crate) fn short_type_name<T: ?Sized>() -> String {
75    let full = std::any::type_name::<T>();
76    let before_generics = full.split('<').next().unwrap_or(full);
77    before_generics
78        .rsplit("::")
79        .next()
80        .unwrap_or(before_generics)
81        .to_string()
82}

84/// Forwarding impl so a `Box<dyn BehaviorNode<D>>` (or any boxed node) is itself
85/// a [`BehaviorNode`]. This lets boxed nodes be used anywhere an
86/// `impl BehaviorNode<D>` is expected.
87impl<D, N: BehaviorNode<D> + ?Sized> BehaviorNode<D> for Box<N> {
88    #[inline]
89    fn tick(&mut self, data: &mut D, dbg: Option<&mut DebugNode>) -> Status {
90        (**self).tick(data, dbg)
91    }
92
93    #[inline]
94    fn halt(&mut self) {
95        (**self).halt()
96    }
97
98    #[inline]
99    fn node_info(&self) -> String {
100        (**self).node_info()
101    }
102
103    #[inline]
104    fn tick_traced(&mut self, data: &mut D) -> (Status, DebugNode) {
105        (**self).tick_traced(data)
106    }
107}

109/// Convenience for turning any concrete node into a [`BoxNode`].
110///
111/// ```
112/// use btree::prelude::*;
113///
114/// let node: BoxNode<()> = AlwaysSuccess.boxed();
115/// ```
116pub trait IntoBoxNode<D> {
117    /// Box this node up as a type-erased [`BoxNode`].
118    fn boxed(self) -> BoxNode<D>;
119}
120
121impl<D, T: BehaviorNode<D> + 'static> IntoBoxNode<D> for T {
122    #[inline]
123    fn boxed(self) -> BoxNode<D> {
124        Box::new(self)
125    }
