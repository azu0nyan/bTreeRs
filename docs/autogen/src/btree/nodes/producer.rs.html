1//! The [`Producer`] node, which builds its child lazily from the context.
2
3use crate::debug::{record, tick_child};
4use crate::{BehaviorNode, BoxNode, DebugNode, Status};
5
6/// A boxed factory closure that builds a child subtree from the context.
7type ProduceFn<D> = Box<dyn FnMut(&mut D) -> BoxNode<D>>;
8
9/// Lazily produces a child subtree from the context, ticks it to completion,
10/// then discards it.
11///
12/// On the first tick (or the first tick after the previous child finished) the
13/// `produce` closure is called to build a fresh child from the current context.
14/// That child is ticked until it succeeds or fails, then dropped so the next
15/// run produces a new one. This is the idiomatic way to choose *what* to do
16/// based on runtime state. Mirrors `ProducerNode`.
17///
18/// ```
19/// use btree::prelude::*;
20///
21/// // Re-targets every time it (re)starts.
22/// let mut p: Producer<i32> = Producer::new(|target: &mut i32| {
23///     let goal = *target;
24///     Action::new(move |cur: &mut i32| {
25///         *cur += 1;
26///         if *cur >= goal { Status::Success } else { Status::Running }
27///     })
28///     .boxed()
29/// });
30/// ```
31pub struct Producer<D> {
32    produce: ProduceFn<D>,
33    label: Option<String>,
34    running_node: Option<BoxNode<D>>,
35}
36
37impl<D> Producer<D> {
38    /// Build an unlabeled producer from a factory closure.
39    pub fn new(produce: impl FnMut(&mut D) -> BoxNode<D> + 'static) -> Self {
40        Self {
41            produce: Box::new(produce),
42            label: None,
43            running_node: None,
44        }
45    }
46
47    /// Build a labeled producer (the label appears in traces).
48    pub fn labeled(label: impl Into<String>, produce: impl FnMut(&mut D) -> BoxNode<D> + 'static) -> Self {
49        Self {
50            produce: Box::new(produce),
51            label: Some(label.into()),
52            running_node: None,
53        }
54    }
55}
56
57impl<D> BehaviorNode<D> for Producer<D> {
58    fn tick(&mut self, data: &mut D, mut dbg: Option<&mut DebugNode>) -> Status {
59        if self.running_node.is_none() {
60            let node = (self.produce)(data);
61            self.running_node = Some(node);
62        }
63        // `as_deref_mut` reborrows the boxed child as `&mut dyn BehaviorNode`
64        // for the duration of the call only, so we can clear `running_node`
65        // in the match arms below.
66        let status = match tick_child(self.running_node.as_deref_mut().unwrap(), data, &mut dbg) {
67            Status::Success => {
68                self.running_node = None;
69                Status::Success
70            }
71            Status::Failure => {
72                self.running_node = None;
73                Status::Failure
74            }
75            Status::Running => Status::Running,
76        };
77        record(dbg, status, || self.node_info());
78        status
79    }
80
81    fn halt(&mut self) {
82        if let Some(node) = self.running_node.as_mut() {
83            node.halt();
84        }
85        self.running_node = None;
86    }
87
88    fn node_info(&self) -> String {
89        let state = if self.running_node.is_some() {
90            "Running"
91        } else {
92            "Halted"
93        };
94        match &self.label {
95            Some(l) => format!("Producer : {l} : {state}"),
96            None => format!("Producer : {state}"),
97        }
98    }
99}
