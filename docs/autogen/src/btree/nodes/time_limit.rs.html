1//! The [`TimeLimit`] decorator.
2
3use crate::caps::HasDelta;
4use crate::debug::{record, tick_child};
5use crate::{BehaviorNode, BoxNode, DebugNode, Status};
6
7/// Fails its child if it runs for longer than `limit` seconds. Mirrors
8/// `BTTimeLimit`.
9///
10/// Elapsed time accumulates from the context's [`HasDelta::delta_seconds`]
11/// while the child runs. If the child finishes within the limit its result is
12/// returned (and the timer resets); if the limit is exceeded first the child is
13/// halted and the node returns [`Status::Failure`].
14pub struct TimeLimit<D> {
15    child: BoxNode<D>,
16    limit: f64,
17    elapsed: f64,
18}
19
20impl<D> TimeLimit<D> {
21    /// Wrap `child` so it fails if it runs longer than `limit` seconds.
22    pub fn new(child: impl BehaviorNode<D> + 'static, limit: f64) -> Self {
23        Self {
24            child: Box::new(child),
25            limit,
26            elapsed: 0.0,
27        }
28    }
29
30    fn reset(&mut self) {
31        self.child.halt();
32        self.elapsed = 0.0;
33    }
34}
35
36impl<D: HasDelta> BehaviorNode<D> for TimeLimit<D> {
37    fn tick(&mut self, data: &mut D, mut dbg: Option<&mut DebugNode>) -> Status {
38        self.elapsed += data.delta_seconds();
39        if self.elapsed > self.limit {
40            self.reset();
41            record(dbg, Status::Failure, || "TimeLimit (expired)".to_string());
42            return Status::Failure;
43        }
44        let status = tick_child(&mut *self.child, data, &mut dbg);
45        if status.is_done() {
46            self.reset();
47        }
48        record(dbg, status, || "TimeLimit".to_string());
49        status
50    }
51
52    fn halt(&mut self) {
53        self.reset();
54    }
55
56    fn node_info(&self) -> String {
57        format!("TimeLimit {:.3} / {:.3}", self.elapsed, self.limit)
58    }
59}
