1//! The [`Predicate`] leaf.
2
3use crate::debug::record;
4use crate::{BehaviorNode, DebugNode, Status};
5
6/// A leaf that succeeds when its predicate is `true` and fails otherwise.
7///
8/// Mirrors the Scala `PredicateNode`. The closure receives the context so it
9/// can query the agent/blackboard; ignore the argument if you only need to
10/// read captured state.
11///
12/// ```
13/// use btree::prelude::*;
14///
15/// let mut hungry = Predicate::labeled("hungry", |hp: &mut i32| *hp < 50);
16/// let mut hp = 30;
17/// assert_eq!(hungry.tick(&mut hp, None), Status::Success);
18/// ```
19pub struct Predicate<D> {
20    predicate: Box<dyn FnMut(&mut D) -> bool>,
21    label: Option<String>,
22    last_result: bool,
23}
24
25impl<D> Predicate<D> {
26    /// Create an unlabeled predicate.
27    pub fn new(predicate: impl FnMut(&mut D) -> bool + 'static) -> Self {
28        Self {
29            predicate: Box::new(predicate),
30            label: None,
31            last_result: true,
32        }
33    }
34
35    /// Create a predicate with a label shown in debug traces.
36    pub fn labeled(label: impl Into<String>, predicate: impl FnMut(&mut D) -> bool + 'static) -> Self {
37        Self {
38            predicate: Box::new(predicate),
39            label: Some(label.into()),
40            last_result: true,
41        }
42    }
43}
44
45impl<D> BehaviorNode<D> for Predicate<D> {
46    fn tick(&mut self, data: &mut D, dbg: Option<&mut DebugNode>) -> Status {
47        self.last_result = (self.predicate)(data);
48        let status = if self.last_result { Status::Success } else { Status::Failure };
49        record(dbg, status, || self.node_info());
50        status
51    }
52
53    fn node_info(&self) -> String {
54        match &self.label {
55            Some(l) => format!("Predicate: {l} : {}", self.last_result),
56            None => format!("Predicate : {}", self.last_result),
57        }
58    }
59}
