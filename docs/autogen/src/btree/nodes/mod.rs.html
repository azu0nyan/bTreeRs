1//! Concrete behavior-tree nodes — one type per submodule.
2//!
3//! These are the building blocks you compose into a tree:
4//!
5//! * **Leaves** that look at and act on the context: [`Predicate`], [`Action`],
6//!   [`RunCustomFunc`], [`AlwaysSuccess`], [`AlwaysFailure`], [`AlwaysRunning`],
7//!   the timers [`Wait`], [`WaitTicks`] and [`RandomWait`], and the navigation
8//!   leaves [`SetDestination`] and [`GoTo`].
9//! * **Decorators** that transform a single child: [`Invert`], [`ForceSuccess`],
10//!   [`ForceFailure`], [`Repeat`], [`Retry`], [`RepeatUntilSuccess`],
11//!   [`RepeatUntilFailure`], [`RunLimit`], [`Delay`], [`Cooldown`],
12//!   [`TimeLimit`], [`Probability`].
13//! * **Composites** that run a list of children: the sequence family
14//!   ([`Sequence`], [`ReactiveSequence`], [`ProgressiveSequence`]), the fallback
15//!   / selector family ([`FallbackSequence`], [`ReactiveFallbackSequence`]),
16//!   [`Parallel`], [`ForEach`], and the randomized
17//!   [`RandomSequence`] / [`RandomSelector`] / [`ProbabilitySelector`].
18//! * **Conditional branches**: [`IfThenElse`] (latching) and
19//!   [`ReactiveIfThenElse`] (re-evaluating).
20//! * The lazy [`Producer`].
21//!
22//! Every type here implements [`BehaviorNode`] over a context type `D`. Some
23//! bound `D` on a capability trait (see [`caps`](crate::caps)): the timers need
24//! [`HasDelta`](crate::HasDelta), the randomized nodes need [`HasRng`], and the
25//! navigation leaves need [`HasNavAgent`](crate::HasNavAgent) /
26//! [`FollowPath`](crate::FollowPath). They are re-exported at the crate root (and from
27//! the [`prelude`](crate::prelude)), so `btree::Sequence` and
28//! `btree::nodes::Sequence` name the same type.
29

120/// Halt every child node. Shared by the sequence / fallback composites.
121fn halt_all<D>(children: &mut [BoxNode<D>]) {
122    for c in children.iter_mut() {
123        c.halt();
124    }
125}

127/// A shuffled `0..len` index order drawn from the context RNG. Shared by the
128/// randomized composites ([`RandomSequence`] / [`RandomSelector`]).
129fn shuffled_order<D: HasRng>(len: usize, data: &mut D) -> Vec<usize> {
130    let mut order: Vec<usize> = (0..len).collect();
131    // Fisher–Yates.
132    for i in (1..len).rev() {
133        let j = data.next_below(i + 1);
134        order.swap(i, j);
135    }
136    order
137}

139/// Which branch of a conditional is currently running. Shared by
140/// [`IfThenElse`] and [`ReactiveIfThenElse`].
141#[derive(Clone, Copy, PartialEq, Eq, Debug)]
142enum Branch {
143    Then,
144    Else,
145}
146
147/// Render a (possibly absent) running branch for `node_info`.
148fn branch_word(running: Option<Branch>) -> &'static str {
149    match running {
150        Some(Branch::Then) => "then",
151        Some(Branch::Else) => "else",
152        None => "NO BRANCH",
153    }
