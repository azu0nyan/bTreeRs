1//! The [`ProbabilitySelector`] composite.
2
3use crate::caps::HasRng;
4use crate::debug::{record, tick_child};
5use crate::{BehaviorNode, BoxNode, DebugNode, Status};
6
7/// Picks one child at random — weighted by per-child probabilities — and runs
8/// it to completion. Mirrors `BTProbabilitySelector`.
9///
10/// On entry a single child is chosen by a weighted draw from the context's
11/// [`HasRng`]; that child is then ticked (across as many ticks as it needs) and
12/// its result returned. Once it finishes the choice is cleared so the next
13/// entry re-draws. If the weights sum to zero the node fails.
14pub struct ProbabilitySelector<D> {
15    weights: Vec<f64>,
16    children: Vec<BoxNode<D>>,
17    chosen: Option<usize>,
18}
19
20impl<D> ProbabilitySelector<D> {
21    /// Build from `(weight, child)` pairs. Weights need not sum to 1; they are
22    /// normalized by their total at draw time. Negative weights are clamped to
23    /// zero.
24    pub fn new(weighted: Vec<(f64, BoxNode<D>)>) -> Self {
25        let mut weights = Vec::with_capacity(weighted.len());
26        let mut children = Vec::with_capacity(weighted.len());
27        for (w, child) in weighted {
28            weights.push(w.max(0.0));
29            children.push(child);
30        }
31        Self { weights, children, chosen: None }
32    }
33}
34
35impl<D: HasRng> BehaviorNode<D> for ProbabilitySelector<D> {
36    fn tick(&mut self, data: &mut D, mut dbg: Option<&mut DebugNode>) -> Status {
37        if self.chosen.is_none() {
38            let total: f64 = self.weights.iter().sum();
39            if total <= 0.0 {
40                record(dbg, Status::Failure, || "ProbabilitySelector (no weight)".to_string());
41                return Status::Failure;
42            }
43            let mut roll = data.next_f64() * total;
44            let mut pick = self.weights.len() - 1;
45            for (i, w) in self.weights.iter().enumerate() {
46                if roll < *w {
47                    pick = i;
48                    break;
49                }
50                roll -= *w;
51            }
52            self.chosen = Some(pick);
53        }
54        let idx = self.chosen.unwrap();
55        let status = tick_child(&mut *self.children[idx], data, &mut dbg);
56        if status.is_done() {
57            self.chosen = None;
58        }
59        record(dbg, status, || self.node_info());
60        status
61    }
62
63    fn halt(&mut self) {
64        if let Some(idx) = self.chosen {
65            self.children[idx].halt();
66        }
67        self.chosen = None;
68    }
69
70    fn node_info(&self) -> String {
71        match self.chosen {
72            Some(i) => format!("ProbabilitySelector -> {i}"),
73            None => "ProbabilitySelector".to_string(),
74        }
75    }
76}
