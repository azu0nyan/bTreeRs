1//! The [`Status`] returned by every behavior-node tick.
2
3/// Result of ticking a [`BehaviorNode`](crate::BehaviorNode).
4///
5/// This mirrors the three-valued result used by virtually every behavior-tree
6/// formulation:
7///
8/// * [`Status::Success`] — the node finished its work successfully.
9/// * [`Status::Failure`] — the node finished, but failed.
10/// * [`Status::Running`] — the node needs more ticks to finish; it should be
11///   ticked again next frame.
12#[derive(Clone, Copy, PartialEq, Eq, Hash, Debug)]
13pub enum Status {
14    /// The node completed successfully.
15    Success,
16    /// The node completed, unsuccessfully.
17    Failure,
18    /// The node is still working and wants to be ticked again.
19    Running,
20}
21
22impl Status {
23    /// `true` for [`Status::Success`].
24    #[inline]
25    pub fn is_success(self) -> bool {
26        matches!(self, Status::Success)
27    }
28
29    /// `true` for [`Status::Failure`].
30    #[inline]
31    pub fn is_failure(self) -> bool {
32        matches!(self, Status::Failure)
33    }
34
35    /// `true` for [`Status::Running`].
36    #[inline]
37    pub fn is_running(self) -> bool {
38        matches!(self, Status::Running)
39    }
40
41    /// `true` when the node has finished this tick, i.e. it is not
42    /// [`Status::Running`].
43    #[inline]
44    pub fn is_done(self) -> bool {
45        !self.is_running()
46    }
47}
