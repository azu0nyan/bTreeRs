1//! The [`Parallel`] composite and its [`ParallelPolicy`].
2
3use super::halt_all;
4use crate::debug::{record, tick_child};
5use crate::{BehaviorNode, BoxNode, DebugNode, Status};
6
7/// When a [`Parallel`] composite considers a success or failure condition met.
8#[derive(Clone, Copy, PartialEq, Eq, Debug)]
9pub enum ParallelPolicy {
10    /// The condition is met only once *every* child has reached it.
11    RequireAll,
12    /// The condition is met as soon as *one* child reaches it.
13    RequireOne,
14}
15
16/// Ticks all of its still-running children every tick, resolving by a pair of
17/// [`ParallelPolicy`] rules. Mirrors `BTParallel`.
18///
19/// Each tick, every child that has not yet finished is ticked once. A child
20/// that returns [`Status::Success`] or [`Status::Failure`] is recorded as done
21/// and is not ticked again until the composite resets. The composite then
22/// resolves:
23///
24/// * **Failure** if the `failure` policy is met (`RequireOne`: at least one
25///   child failed; `RequireAll`: all children failed), else
26/// * **Success** if the `success` policy is met (`RequireOne`: at least one
27///   child succeeded; `RequireAll`: all children succeeded), else
28/// * [`Status::Running`].
29///
30/// On resolving it halts every child and clears its bookkeeping, so the next
31/// entry runs all children afresh. [`Parallel::new`] uses the common "all must
32/// succeed, any failure aborts" configuration.
33pub struct Parallel<D> {
34    children: Vec<BoxNode<D>>,
35    results: Vec<Option<Status>>,
36    success: ParallelPolicy,
37    failure: ParallelPolicy,
38}
39
40impl<D> Parallel<D> {
41    /// Build a parallel that succeeds when all children succeed and fails as
42    /// soon as one child fails.
43    pub fn new(children: Vec<BoxNode<D>>) -> Self {
44        Self::with_policies(children, ParallelPolicy::RequireAll, ParallelPolicy::RequireOne)
45    }
46
47    /// Build a parallel with explicit `success` and `failure` policies.
48    pub fn with_policies(children: Vec<BoxNode<D>>, success: ParallelPolicy, failure: ParallelPolicy) -> Self {
49        let results = vec![None; children.len()];
50        Self {
51            children,
52            results,
53            success,
54            failure,
55        }
56    }
57
58    fn reset(&mut self) {
59        halt_all(&mut self.children);
60        for r in self.results.iter_mut() {
61            *r = None;
62        }
63    }
64}
65
66impl<D> BehaviorNode<D> for Parallel<D> {
67    fn tick(&mut self, data: &mut D, mut dbg: Option<&mut DebugNode>) -> Status {
68        let len = self.children.len();
69        for i in 0..len {
70            if self.results[i].is_none() {
71                let s = tick_child(&mut *self.children[i], data, &mut dbg);
72                if s.is_done() {
73                    self.results[i] = Some(s);
74                }
75            }
76        }
77
78        let successes = self.results.iter().filter(|r| **r == Some(Status::Success)).count();
79        let failures = self.results.iter().filter(|r| **r == Some(Status::Failure)).count();
80
81        let failed = match self.failure {
82            ParallelPolicy::RequireOne => failures >= 1,
83            ParallelPolicy::RequireAll => failures == len,
84        };
85        let succeeded = match self.success {
86            ParallelPolicy::RequireOne => successes >= 1,
87            ParallelPolicy::RequireAll => successes == len,
88        };
89
90        let status = if failed {
91            self.reset();
92            Status::Failure
93        } else if succeeded {
94            self.reset();
95            Status::Success
96        } else {
97            Status::Running
98        };
99        record(dbg, status, || self.node_info());
100        status
101    }
102
103    fn halt(&mut self) {
104        self.reset();
105    }
106
107    fn node_info(&self) -> String {
108        let done = self.results.iter().filter(|r| r.is_some()).count();
109        format!("Parallel {} / {}", done, self.children.len())
110    }
111}
